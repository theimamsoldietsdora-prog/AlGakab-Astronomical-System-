import math from datetime import datetime, timedelta # ============================================================================== # 1. الثوابت الأساسية وهيكل التقويم المعيني (System Constants & Calendar Structure) # ============================================================================== # معامل التمدد المعيني الثابت C_M = 0.03105 # مرجع انطلاق التقويم المعيني (11 ديسمبر - أول شهر رمضان المعيني) ANCHOR_MONTH = 12 ANCHOR_DAY = 11 # أسماء الأشهر المعينية بالترتيب MEEANI_MONTH_NAMES = [ "1. رمضان المبارك", "2. شوال", "3. ذو القعدة", "4. ذو الحجة", "5. محرم", "6. صفر", "7. ربيع الأول (القمة)", "8. ربيع الثاني", "9. جمادى الأولى", "10. جمادى الآخرة", "11. رجب", "12. شعبان" ] # نمط الساعات المضافة للأشهر (أيام الأساس = 29 ما عدا الشهر 7 = 30) HOURLY_PATTERN = [0, 1, 2, 10, 15, 21, 0, 23, 21, 16, 12, 6] # ============================================================================== # 2. محرك حساب التقويم المعيني المطاط (Me'eeni Calendar Engine) # ============================================================================== def get_meeani_calendar_structure(): """ يقوم بحساب الهيكل الكامل للأشهر الـ 12 بالقيم الأساسية ونسب الزيادة المعينية 0.03105 والقيم الكلية النهائية بالأيام. """ calendar_data = [] total_year_days = 0.0 for i in range(12): base_days = 30.0 if i == 6 else 29.0 # ربيع الأول هو القمة (30 يوم) added_hours = HOURLY_PATTERN[i] # القيمة بالأيام الستينية قبل الضرب raw_days = base_days + (added_hours / 24.0) # الزيادة المعينية بضرب النسبة الثابتة 0.03105 meeani_addition = raw_days * C_M # القيمة الكلية النهائية للشهر المطاطي final_elastic_days = raw_days + meeani_addition total_year_days += final_elastic_days calendar_data.append({ "month_num": i + 1, "name": MEEANI_MONTH_NAMES[i], "base_days": base_days, "added_hours": added_hours, "raw_days": raw_days, "addition": meeani_addition, "final_days": final_elastic_days }) return calendar_data, total_year_days # ============================================================================== # 3. التزامن الفلكي التلقائي ومؤشر اليوم الحالي (Automatic Anchor & Live Tracking) # ============================================================================== def locate_current_meeani_day(current_datetime, calendar_structure): """ يبحث أوتوماتيكياً عن التاريخ الفعلي الحالي مقارنة بنقطة الارتكاز (11/12) ويحدد في أي شهر ويوم معيني تقع الآن بدقة مطاطية. """ # تحديد بداية السنة المعينية الحالية من 11 ديسمبر الماضي year = current_datetime.year anchor_date = datetime(year, ANCHOR_MONTH, ANCHOR_DAY) # إذا كان التاريخ الحالي قبل 11 ديسمبر، فنحن في سنة معينية بدأت 11 ديسمبر العام الماضي if current_datetime < anchor_date: anchor_date = datetime(year - 1, ANCHOR_MONTH, ANCHOR_DAY) # عدد الأيام الفردية المنقضية منذ 11/12 days_passed = (current_datetime - anchor_date).total_seconds() / 86400.0 accumulated_days = 0.0 current_month_info = None day_in_month = 0 for month in calendar_structure: month_length = month["final_days"] if accumulated_days <= days_passed < (accumulated_days + month_length): current_month_info = month day_in_month = int(days_passed - accumulated_days) + 1 break accumulated_days += month_length # في حال تجاوزت السنة if not current_month_info: current_month_info = calendar_structure[-1] day_in_month = int(calendar_structure[-1]["final_days"]) return anchor_date, current_month_info, day_in_month, days_passed # ============================================================================== # 4. محرك الساعة المطاطية المتزامنة مع الشروق (Sunrise-Anchored Elastic Clock) # ============================================================================== def calculate_elastic_clock(current_datetime, sunrise_time_str="06:30:00"): """ يزامن أول ساعة في النهار مع لحظة شروق الشمس تماماً لليوم الحالي (بداية صفر الوقت) ويقوم بمط الوقت وحساب الساعة والدقيقة المعينية المطاطية. """ # تحويل وقت الشروق إلى كائن datetime لليوم الحالي sunrise_time = datetime.strptime( f"{current_datetime.strftime('%Y-%m-%d')} {sunrise_time_str}", "%Y-%m-%d %H:%M:%S" ) # إذا كان الوقت الحالي قبل الشروق، نأخذ شروق اليوم السابق كمرجع if current_datetime < sunrise_time: sunrise_time -= timedelta(days=1) # الثواني الستينية المنقضية منذ الشروق elapsed_seconds = (current_datetime - sunrise_time).total_seconds() elapsed_minutes = elapsed_seconds / 60.0 # تحويل الدقائق الستينية إلى دقائق معينية مطاطية بواسطة النسبة الثابتة 0.03105 elastic_minutes_passed = elapsed_minutes * (1 + C_M) # حساب الساعات والدقائق والثواني المطاطية (تعتمد 60 دقيقة مطاطية للساعة) elastic_hours = int(elastic_minutes_passed // 60) remaining_elastic_minutes = int(elastic_minutes_passed % 60) elastic_seconds = int((elastic_minutes_passed * 60) % 60) return { "sunrise_anchor": sunrise_time.strftime("%H:%M:%S"), "elapsed_standard_mins": elapsed_minutes, "elastic_hours": elastic_hours, "elastic_minutes": remaining_elastic_minutes, "elastic_seconds": elastic_seconds } # ============================================================================== # 5. عرض الهيكل بالكامل واجهة التقويم والساعة (Full Display Interface) # ============================================================================== def display_full_meeani_system(): # 1. بناء هيكل التقويم calendar_structure, total_year_days = get_meeani_calendar_structure() # 2. الحصول على التاريخ الحالي وتحديد المؤشر الأوتوماتيكي now = datetime.now() anchor_date, current_month, day_in_month, days_passed = locate_current_meeani_day(now, calendar_structure) # 3. حساب الساعة المطاطية المزامنة مع الشروق (افتراض شروق بغداد اليوم 06:15 ص) clock_info = calculate_elastic_clock(now, sunrise_time_str="06:15:00") print("=" * 80) print(" نظام التقويم والساعة المطاطية المعينية") print("=" * 80) print(f"النسبة المعينية الثابتة (C_m): {C_M}") print(f"مرجع التقويم (بداية 1 رمضان): {anchor_date.strftime('%Y-%m-%d')} (11 ديسمبر)") print(f"تاريخ النظام الحالي: {now.strftime('%Y-%m-%d %H:%M:%S')}") print("-" * 80) # عرض الساعة المطاطية أعلاه print(" [ الساعة المعينية المطاطية الآن (بداية التوقيت من الشروق) ]") print(f" • نقطة بداية نهار اليوم (شروق الشمس): {clock_info['sunrise_anchor']}") print(f" • الوقت المطاط الآن: {clock_info['elastic_hours']:02d}:{clock_info['elastic_minutes']:02d}:{clock_info['elastic_seconds']:02d} (ساعة مطاطية)") print("-" * 80) # عرض الجدول الكامل للأشهر print(" [ هيكل التقويم المعيني للسنة الكاملة ]") print(f"{'#':<3} | {'اسم الشهر المعيني':<22} | {'الأساس (أيام+س)':<15} | {'الزيادة (0.03105)':<15} | {'إجمالي أيام الشهر المطاطية':<20}") print("-" * 80) for m in calendar_structure: base_fmt = f"{int(m['base_days'])}d + {m['added_hours']}h" print(f"{m['month_num']:<3} | {m['name']:<22} | {base_fmt:<15} | +{m['addition']:<14.5f} | {m['final_days']:<20.5f}") print("-" * 80) print(f" المجموع الإجمالي للسنة المعينية المطاطية: {total_year_days:.5f} يوماً") print("=" * 80) # عرض مؤشر اليوم الحالي الأوتوماتيكي print(" [ مؤشر موقعك الفعلي الآن في التقويم المعيني (أوتوماتيكي) ]") print(f" • أنت اليوم في: {current_month['name']}") print(f" • اليوم المعيني: {day_in_month} من أصل {current_month['final_days']:.2f} يوماً مطاطياً") print(f" • الأيام الكلية المنقضية منذ 11/12: {days_passed:.2f} يوماً") print("=" * 80) # تشغيل النظام بالكامل if __name__ == "__main__": display_full_meeani_system()
