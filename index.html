import math from datetime import datetime, timedelta class MeeeniCalendarEngine: def __init__(self, current_date=None, latitude_deg=33.0): """ محرك التقويم والزمن المطاطي المعيني الشامل (Me'eeni Unified Engine) المرجع الأساسي: 10 كانون الأول (ديسمبر) كبداية للسنة المعينية. """ self.latitude_deg = latitude_deg self.days_in_year = 365.2425 # تحديد التاريخ الحالي (افتراضياً اليوم الحقيقي أو تاريخ محاذاة) self.current_real_date = current_date if current_date else datetime.now() def get_seasonal_amplitude(self): """معالج المعايرة الجغرافية وخطوط العرض (Geographic Calibration Processor)""" base_delta = 10 * math.sin(math.radians(23.5)) lat_factor = math.sin(math.radians(self.latitude_deg)) / math.sin(math.radians(33.0)) return base_delta * lat_factor def calculate_fowaqa(self, day_of_year): """حساب الفواقة النهارية والليلية لليوم (t) ضمن مسار الأنفينيتي""" delta = self.get_seasonal_amplitude() angle = (2 * math.pi * day_of_year) / self.days_in_year # الفواقة النهارية (تنكمش في الأوج شتاءً بـ 47 درجة، وتتمدد في الحضيض صيفاً) f_day = 60.0 - delta * math.cos(angle) # الفواقة الليلية المتعاكسة لضمان بقاء كتلة اليوم 24 ساعة تامة f_night = 60.0 + delta * math.cos(angle) return f_day, f_night def generate_meeeni_calendar(self): """ توليد تقويم السنة المعينية بالكامل ابتداءً من 10 كانون الأول (ديسمبر) مع أسماء الأشهر المعينية، عدد الأيام (29 أو 30 للقمر)، ومؤشر اليوم الحي. """ # بناء أشهر السنة المعينية الـ 12 متوافقة مع الدورة القمرية وحركة المحور Z # إجمالي الأيام = 365 يوماً (موزعة على 12 شهراً تتراوح بين 29 و 30 يوماً) months_structure = [ ("كانون الأول (شهر التأسيس والأوج)", 30), ("كانون الثاني", 30), ("شباط", 29), ("آذار (الاعتدال الربيعي)", 30), ("نيسان", 29), ("أيار", 30), ("حزيران (الحضيض والتعامد الصيفي)", 30), ("تموز", 29), ("آب", 30), ("أيلول (الاعتدال الخريفي)", 29), ("تشرين الأول", 30), ("تشرين الثاني", 29) ] # نقطة البداية المرجعية للكتاب: 10 كانون الأول من العام الحالي/السابق anchor_dec_10 = datetime(self.current_real_date.year if self.current_real_date.month >= 12 else self.current_real_date.year - 1, 12, 10) calendar_db = [] global_day_counter = 0 for m_idx, (m_name, m_days) in enumerate(months_structure): month_days_list = [] for d_in_m in range(1, m_days + 1): # التاريخ الميلادي المقابل لكل يوم معيني current_gregorian_date = anchor_dec_10 + timedelta(days=global_day_counter) # حساب الفواقات واللمحات لهذا اليوم المعيني بالتحديد f_day, f_night = self.calculate_fowaqa(global_day_counter) l_day = f_day / 60.0 # اللمحة النهارية l_night = f_night / 60.0 # اللمحة الليلية # التحقق من إثبات كتلة اليوم (86,400 ثانية ستينية = 24 ساعة معينية) total_seconds = (12 * 60 * f_day) + (12 * 60 * f_night) # هل هو اليوم الحي الذي يعيشه المستخدم الآن؟ is_today = (current_gregorian_date.date() == self.current_real_date.date()) day_info = { "day_index_in_year": global_day_counter, "gregorian_date": current_gregorian_date.strftime("%Y-%m-%d"), "day_name_in_month": f"اليوم {d_in_m} من {m_name}", "f_day": round(f_day, 3), "f_night": round(f_night, 3), "l_day": round(l_day, 3), "total_mass_seconds": round(total_seconds, 1), "is_live_today": is_today } month_days_list.append(day_info) global_day_counter += 1 calendar_db.append({ "month_name": m_name, "days_count": m_days, "days": month_days_list }) return calendar_db, anchor_dec_10 def display_live_dashboard(self): """عرض لوحة التحكم الحية للتقويم والوقت المطاطي المعيني""" cal, start_anchor = self.generate_meeeni_calendar() print("==================================================================") print(" محرك تقويم الزمن المطاطي المعيني (Me'eeni Engine) ") print(f" نقطة الارتكاز الكونية: 10 كانون الأول ({start_anchor.strftime('%Y')}) ") print("==================================================================") live_found = False for month in cal: print(f"\n[ الشهر المعيني: {month['month_name']} ] (عدد الأيام: {month['days_count']})") print("-" * 65) print(f"{'التاريخ الميلادي':<12} | {'اليوم المعيني':<22} | {'F_day (نهار)':<12} | {'F_night (ليل)':<12}") print("-" * 65) for day in month['days']: marker = " ==> [أنت هنا اليوم]" if day['is_live_today'] else "" if day['is_live_today']: live_found = True print(f"{day['gregorian_date']:<12} | {day['day_name_in_month']:<22} | {day['f_day']:<12} | {day['f_night']:<12}{marker}") if not live_found: print("\n[ملاحظة]: التاريخ الفعلي اليوم يقع خارج نطاق هذه السنة المعينية الحالية المعروضة، يتم ضبط المؤشر تلقائياً.") # --- التشغيل الفعلي للمحرك وتقويم السنة --- if __name__ == "__main__": # تشغيل النظام بناءً على الوقت الحقيقي الحالي engine = MeeeniCalendarEngine(latitude_deg=33.0) engine.display_live_dashboard()
